cmake_minimum_required(VERSION 3.25)
project(BlackWhiteDecomp VERSION 0.11.1 LANGUAGES CXX RC)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Build with MSVC for 32-bit x86 (-A Win32): the code runs inside the 32-bit runblack.exe.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Static CRT: nothing to install next to the game. The game's own CRT and heap are separate (see core/GameHeap.h).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(bw_options INTERFACE)
target_compile_options(bw_options INTERFACE
    "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/utf-8;/Zc:preprocessor;/Zc:__cplusplus;/EHsc;/arch:SSE2>")
target_compile_definitions(bw_options INTERFACE
    _WIN32_WINNT=0x0A00 WINVER=0x0A00 UNICODE _UNICODE)
target_include_directories(bw_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_BINARY_DIR}/generated)

# The patch's own version (CHANGELOG.md), logged at start-up and shown by the launcher.
configure_file(src/core/Version.h.in generated/core/Version.h @ONLY)

# A Windows version resource (file properties) for `target`. `type` is VFT_APP or VFT_DLL.
function(bw_version_resource target file_name description type)
    set(BW_FILE_NAME "${file_name}")
    set(BW_FILE_DESCRIPTION "${description}")
    set(BW_FILE_TYPE "${type}")
    configure_file(src/core/Version.rc.in generated/version/${target}.rc @ONLY)
    target_sources(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/version/${target}.rc)
endfunction()

# Infrastructure shared by the DLL, the launcher and the tests.
add_library(bw_core STATIC
    src/core/Bitmap.cpp
    src/core/Config.cpp
    src/core/GameExecutable.cpp
    src/core/GameHeap.cpp
    src/core/Hook.cpp
    src/core/Image.cpp
    src/core/Log.cpp
    src/core/Patch.cpp)
target_link_libraries(bw_core PUBLIC bw_options bcrypt windowscodecs)

# Decompiled game code and fixes. An OBJECT library so every translation unit (and its hook registrations) is linked
# even though nothing references it by name.
add_library(bw_game OBJECT
    src/game/black/CPUCheck.cpp
    src/game/black/PCMain.cpp
    src/game/lh3dlib/LH3DRender.cpp
    src/game/lh3dlib/LH3DSky.cpp
    src/game/lh3dlib/LH3DTexture.cpp
    src/game/lhlib/LHMouse.cpp
    src/game/lhlib/LHScreen.cpp
    src/game/lhlib/LHWin.cpp
    src/diagnostics/Diagnostics.cpp
    src/diagnostics/DirectXTrace.cpp
    src/diagnostics/Screenshot.cpp
    src/render/Com.cpp
    src/render/DepthView.cpp
    src/render/Device.cpp
    src/render/DirectDraw.cpp
    src/render/Format.cpp
    src/render/Gpu.cpp
    src/render/Pixels.cpp
    src/render/Renderer.cpp
    src/render/Surface.cpp
    src/render/TexturePack.cpp
    src/render/Translate.cpp
    src/fixes/Fixes.cpp)
target_link_libraries(bw_game PUBLIC bw_core ddraw dxguid d3d9)

add_library(BWDecomp SHARED src/dll/DllMain.cpp)
target_link_libraries(BWDecomp PRIVATE bw_game)
bw_version_resource(BWDecomp BWDecomp.dll "Black & White patch: decompiled code, fixes and renderer" VFT_DLL)

add_executable(BWLaunch WIN32 src/launcher/Launcher.cpp)
target_link_libraries(BWLaunch PRIVATE bw_core shell32)
bw_version_resource(BWLaunch BWLaunch.exe "Black & White patch launcher" VFT_APP)

# The setup program (see package.ps1): BWLaunch.exe, BWDecomp.dll, the default settings and the player documentation
# as resources of one executable, which installs them into a game folder and doubles as the uninstaller.
configure_file(release/README.txt.in generated/release/README.txt @ONLY NEWLINE_STYLE CRLF)
set(BW_PAYLOAD_RC ${CMAKE_CURRENT_BINARY_DIR}/generated/setup/$<CONFIG>/Payload.rc)
file(GENERATE OUTPUT ${BW_PAYLOAD_RC} CONTENT "// Generated by CMakeLists.txt: the files BWDecompSetup.exe installs (src/setup/Payload.cpp says where).
#include \"setup/Resource.h\"
IDR_PAYLOAD_LAUNCHER RCDATA \"$<TARGET_FILE:BWLaunch>\"
IDR_PAYLOAD_DLL RCDATA \"$<TARGET_FILE:BWDecomp>\"
IDR_PAYLOAD_INI RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/BWDecomp.ini\"
IDR_PAYLOAD_README RCDATA \"${CMAKE_CURRENT_BINARY_DIR}/generated/release/README.txt\"
IDR_PAYLOAD_CHANGELOG RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/CHANGELOG.md\"
IDR_PAYLOAD_FIXES RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/docs/FIXES.md\"
IDR_PAYLOAD_RENDERER RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/docs/RENDERER.md\"
IDR_PAYLOAD_TEXTURES RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/docs/TEXTURES.md\"
")

add_library(bw_setup STATIC src/setup/Install.cpp src/setup/System.cpp)
target_link_libraries(bw_setup PUBLIC bw_core shell32 ole32 advapi32)

add_executable(BWDecompSetup WIN32
    src/setup/Setup.cpp src/setup/Payload.cpp src/setup/Setup.rc src/setup/Setup.manifest ${BW_PAYLOAD_RC})
target_link_libraries(BWDecompSetup PRIVATE bw_setup shlwapi comctl32)
# The manifest above says asInvoker; the linker's own UAC entry would duplicate it.
target_link_options(BWDecompSetup PRIVATE /MANIFESTUAC:NO)
add_dependencies(BWDecompSetup BWLaunch BWDecomp)
bw_version_resource(BWDecompSetup BWDecompSetup.exe "Black & White patch setup" VFT_APP)

enable_testing()
add_executable(bw_tests
    tests/TestMain.cpp tests/CoreTests.cpp tests/GameTests.cpp tests/RenderTests.cpp tests/SetupTests.cpp)
target_link_libraries(bw_tests PRIVATE bw_game bw_setup)
add_test(NAME unit COMMAND bw_tests)
